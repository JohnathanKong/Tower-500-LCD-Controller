cmake_minimum_required(VERSION 3.22)
project(Thermaltake500LCD LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Thermaltake500LCD is a native Windows application")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(lcd_core STATIC src/core.cpp)
target_include_directories(lcd_core PUBLIC "include")
target_compile_definitions(lcd_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(lcd_core PUBLIC gdiplus setupapi hid comctl32 comdlg32 shell32 advapi32 ole32 user32 gdi32)
if(MSVC)
    target_compile_options(lcd_core PRIVATE /W4 /EHsc /utf-8)
endif()
add_executable(lcd_gui WIN32 src/gui.cpp)
target_link_libraries(lcd_gui PRIVATE lcd_core)
set_target_properties(lcd_gui PROPERTIES OUTPUT_NAME "thermaltake500lcd")
add_executable(lcd_worker WIN32 src/service.cpp)
target_link_libraries(lcd_worker PRIVATE lcd_core)
set_target_properties(lcd_worker PROPERTIES OUTPUT_NAME "thermaltake500lcd.service")
install(TARGETS lcd_gui lcd_worker RUNTIME DESTINATION .)
